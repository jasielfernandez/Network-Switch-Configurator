# Network Switch Configuration Commands
# Lines starting with # are comments and will be ignored
# These commands will be executed in order
#
# VARIABLE SUBSTITUTION:
# Use {column_name} to insert values from the CSV file
# Example: {switchname}, {location}, {mgmt_vlan}, etc.
# Variables are substituted per switch from the switches.csv columns
#
# CONDITIONAL LOGIC:
# Use directives to control command execution based on CSV values
#
# IF/ELSE/ENDIF - Conditional blocks:
#   #IF {column} == value
#     commands here
#   #ENDIF
#
#   #IF {vendor} == cisco
#     cisco-specific commands
#   #ELSE
#     commands for other vendors
#   #ENDIF
#
# FOR/ENDFOR - Loop blocks:
#   #FOR variable_name IN {column_or_number}
#     commands with {variable_name}
#   #ENDFOR
#
#   #FOR i IN {stack}
#     interface 1/{i}/48
#   #ENDFOR
#
# Operators: ==, !=, >, <, >=, <=
# Logical: AND, OR
#
# Example:
#   #IF {vendor} == cisco AND {stack} > 0
#     #FOR member IN {stack}
#       interface 1/{member}/1
#       description Stack member {member}
#     #ENDFOR
#   #ENDIF

# Example using variables from CSV
snmp-server location {location}
snmp-server contact Network Team

# Example VLAN configuration using mgmt_vlan variable
vlan {mgmt_vlan}
name Management_VLAN
exit

# Example interface configuration
interface GigabitEthernet1/0/1
description Access Port at {location}
switchport mode access
switchport access vlan {mgmt_vlan}
spanning-tree portfast
exit

# Example logging configuration
logging buffered 16384
logging console informational
